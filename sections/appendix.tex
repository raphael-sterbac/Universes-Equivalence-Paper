\section{Detailed rules}\label{appendix:rules}

All systems share the same rules for context well-formation and context conversion:

\begin{mathparsmall}
  \ebrule{
    \hypo{}
    \infer1{1 \vdash}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdash}
    \hypo{\Gamma \vdash A \;\type}
    \infer2{\Gamma.A \vdash}
  }
  \and
  \ebrule{
    \hypo{}
    \infer1{\vdash 1 = 1}
  }
  \and
  \ebrule{
    \hypo{\vdash \Gamma = \Delta}
    \hypo{\Gamma \vdash A = B}
    \infer2{\vdash \Gamma.A = \Delta.B}
  }
\end{mathparsmall}

\subsection{The system \ThyTarski : Tarski universes}

\begin{mathparsmall}
  \ebrule{
    \hypo{\Gamma \vdashT A \;\type}
    \infer1{\Gamma.A \vdashT v_0 : A\p}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT v_n : A}
    \hypo{\Gamma \vdashT B \;\type}
    \infer2{\Gamma.B \vdashT v_{n+1} : A\p}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT A \;\type}
    \hypo{\Gamma.A \vdashT B \;\type}
    \infer2{\Gamma \vdashT \Pi(A, B) \;\type}
  }
  \and
  \ebrule{
    \hypo{\Gamma.A \vdashT b : B}
    \hypo{\Gamma.A \vdashT B \;\type}
    \hypo{\Gamma \vdashT A \;\type}
    \infer3{\Gamma \vdashT \lambda(A,B,b) : \Pi(A, B)}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT c : \Pi(A, B)}
    \hypo{\Gamma \vdashT a : A}
    \hypo{\Gamma.A \vdashT B \;\type}
    \hypo{\Gamma \vdashT A \;\type}
    \infer4{\Gamma \vdashT \app(A,B,c,a) : B[a]}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT}
    \infer1{\Gamma \vdashT \U_n \;\type}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT a : \U_n}
    \infer1{\Gamma \vdashT T_n(a) \;\type}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT a : \U_n}
    \hypo{\Gamma.T_n(a) \vdashT b : \U_n}
    \hypo{\Gamma \vdashT T_n(a) \;\type}
    \infer3{\Gamma \vdashT \Pi^n(a, b) : \U_n}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT}
    \infer1[($m<n$)]{\Gamma \vdashT \U_m^n : \U_n}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT a : \U_m}
    \infer1[($m\leq n$)]{\Gamma \vdashT t_m^n(a) : \U_n}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT a : A}
    \hypo{\Gamma \vdashT A = B}
    \infer2{\Gamma \vdashT a : B}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT a = a' : A}
    \hypo{\Gamma \vdashT A = B}
    \infer2{\Gamma \vdashT a = a' : B}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT A \;\type}
    \infer1{\Gamma \vdashT A = A}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT A = B}
    \hypo{\Gamma \vdashT A = C}
    \infer2{\Gamma \vdashT B = C}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT a : A}
    \infer1{\Gamma \vdashT a = a : A}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT a = b : A}
    \hypo{\Gamma \vdashT a = c : A}
    \infer2{\Gamma \vdashT b = c : A}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT A = A'}
    \hypo{\Gamma.A \vdashT B = B'}
    \infer2{\Gamma \vdashT \Pi(A, B) = \Pi(A', B')}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT A = A'}
    \hypo{\Gamma.A \vdashT B = B'}
    \hypo{\Gamma.A \vdashT b = b' : B}
    \infer3{\Gamma \vdashT \lambda(A,B,b) = \lambda(A',B',b') : \Pi(A, B)}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT A = A'}
    \hypo{\Gamma.A \vdashT B = B'}
    \hypo{\Gamma \vdashT c = c' : \Pi(A, B)}
    \hypo{\Gamma \vdashT a = a' : A}
    \infer4{\Gamma \vdashT \app(A,B,c,a) = \app(A',B',c',a') : B[a]}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT a = a' : \U_n}
    \infer1{\Gamma \vdashT T_n(a) = T_n(a')}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT a = a' : \U_n}
    \infer1[($n\leq p$)]{\Gamma \vdashT t_n^p(a) = t_n^p(a') : \U_p}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT a = a' : \U_n}
    \hypo{\Gamma.T_n(a) \vdashT b = b' : \U_n}
    \infer2{\Gamma \vdashT \Pi^n(a, b) = \Pi^n(a', b') : \U_n}
  }
  \and
  \ebrule{
    \hypo{\Gamma.A \vdashT b : B}
    \hypo{\Gamma \vdashT a : A}
    \infer2[($\beta$-rule)]{\Gamma \vdashT \app(A,B,\lambda(A,B,b),a) = b[a] : B[a]}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT f : \Pi(A, B)}
    \infer1[($\eta$-rule)]{\Gamma \vdashT f = \lambda(A,B,\app(A\p,B\p^+,f\p, v_0)) : \Pi(A, B)}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT a : \U_n}
    \hypo{\Gamma.T_n(a) \vdashT b : \U_n}
    \infer2{\Gamma \vdashT T_n(\Pi^n(a, b)) = \Pi(T_n(a),T_n(b))}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT a : \U_n}
    \hypo{\Gamma.T_n(a) \vdashT b : \U_n}
    \infer2[($n\leq p$)]{\Gamma \vdashT t_n^p(\Pi^n(a, b)) = \Pi^p(t_n^p(a),t_n^p(b)) : \U_p}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT}
    \infer1[($m<n$)]{\Gamma \vdashT T_n(\U_m^n) = \U_m}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT}
    \infer1[($m<n\leq p$)]{\Gamma \vdashT t_n^p(\U_m^n) = \U_m^p : \U_p}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT a : \U_m}
    \infer1[($m \leq n$)]{\Gamma \vdashT T_n(t_m^n(a)) = T_m(a)}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT a : \U_m}
    \infer1[($m\leq n\leq p$)]{\Gamma \vdashT t_n^p(t_m^n(a)) = t_m^p(a) : \U_p}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashT a : \U_n}
    \infer1{\Gamma \vdashT t_n^n(a) = a : \U_n}
  }
\end{mathparsmall}
