\section{Detailed rules}\label{appendix:rules}

All four systems share the same rules for context well-formation and context conversion:

\begin{mathparsmall}
  \ebrule{
    \hypo{}
    \infer1{1 \vdash}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdash}
    \hypo{\Gamma \vdash A \;\type}
    \infer2{\Gamma.A \vdash}
  }
  \and
  \ebrule{
    \hypo{}
    \infer1{\vdash 1 = 1}
  }
  \and
  \ebrule{
    \hypo{\vdash \Gamma = \Delta}
    \hypo{\Gamma \vdash A = B}
    \infer2{\vdash \Gamma.A = \Delta.B}
  }
\end{mathparsmall}


\subsection{The system \ThyGAT : explicit substitutions}

The system \ThyGAT{} is the deductive presentation of the Generalized Algebraic Theory of Section 2.1, where substitutions are explicit syntactic objects rather than meta-operations. Its raw syntax is given by:
\begin{gather*}
    \Gamma,\Delta ::= 1 \; | \; \Gamma.A\\
    A,B ::= \Pi(A, B) \;|\; (\U_n)_\Gamma \; | \; T_n(a) \; | \; A \sigma\\
    a,b ::= \app(A,B,c,a) \; | \; \lambda(A,B,b) \; | \; t_m^n(a) \; | \; (\U_m^n)_\Gamma \; | \; \Pi^n(a, b) \; | \; \q_{\Gamma,A} \; | \; a\sigma \\
    \sigma , \delta ::= \sigma \delta \; | \; \Id_\Gamma \; | \; ()_\Gamma \; | \; (\sigma, a) \; | \; \p_{\Gamma,A}
\end{gather*}

The formation, equivalence, and congruence rules for substitutions and their actions on types and terms are:

\begin{mathparsmall}
  \ebrule{
    \hypo{\Gamma \vdashG}
    \infer1{\vdashG ()_\Gamma : \Gamma \rightarrow 1}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashG}
    \infer1{\vdashG \Id_\Gamma : \Gamma \rightarrow \Gamma}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashG A \; \type}
    \infer1{\vdashG \p_{\Gamma,A} : \Gamma.A \rightarrow \Gamma}
  }
  \and
  \ebrule{
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \hypo{\Gamma \vdashG A \; \type}
    \hypo{\Delta \vdashG a : A\sigma}
    \infer3{\vdashG (\sigma, a) : \Delta \rightarrow \Gamma.A}
  }
  \and
  \ebrule{
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \hypo{\vdashG \delta : \Theta \rightarrow \Delta}
    \infer2{\vdashG \sigma \delta : \Theta \rightarrow \Gamma}
  }
  \and
  \ebrule{
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \hypo{\Gamma \vdashG A \; \type}
    \infer2{\Delta \vdashG A \sigma \; \type}
  }
  \and
  \ebrule{
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \hypo{\Gamma \vdashG a : A}
    \infer2{\Delta \vdashG a \sigma : A \sigma}
  }
  \and
  \ebrule{
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \infer1{\vdashG \sigma = \sigma : \Delta \rightarrow \Gamma}
  }
  \and
  \ebrule{
    \hypo{\vdashG \sigma = \delta : \Delta \rightarrow \Gamma}
    \hypo{\vdashG \sigma = \nu : \Delta \rightarrow \Gamma}
    \infer2{\vdashG \delta = \nu : \Delta \rightarrow \Gamma}
  }
  \and
  \ebrule{
    \hypo{\vdashG \sigma = \sigma' : \Delta \rightarrow \Gamma}
    \hypo{\vdashG \delta = \delta' : \Theta \rightarrow \Delta}
    \infer2{\vdashG \sigma\delta = \sigma'\delta' : \Theta \rightarrow \Gamma}
  }
  \and
  \ebrule{
    \hypo{\vdashG \sigma = \sigma' : \Delta \rightarrow \Gamma}
    \hypo{\Gamma \vdashG A \; \type}
    \hypo{\Delta \vdashG a = a' : A\sigma}
    \infer3{\vdashG (\sigma, a) = (\sigma', a') : \Delta \rightarrow \Gamma.A}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashG A = A'}
    \hypo{\vdashG \sigma = \sigma' : \Delta \rightarrow \Gamma}
    \infer2{\Delta \vdashG A\sigma = A'\sigma'}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashG a = a' : A}
    \hypo{\vdashG \sigma = \sigma' : \Delta \rightarrow \Gamma}
    \infer2{\Delta \vdashG a\sigma = a'\sigma' : A\sigma}
  }
\end{mathparsmall}

Finally, the conversion rules for the substitution calculus and its naturality on type and term constructors (Definitions 2--7) are:

\begin{mathparsmall}
  \ebrule{
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \infer1{\vdashG \sigma \Id_\Delta = \sigma : \Delta \rightarrow \Gamma}
  }
  \and
  \ebrule{
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \infer1{\vdashG \Id_\Gamma \sigma = \sigma : \Delta \rightarrow \Gamma}
  }
  \and
  \ebrule{
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \hypo{\vdashG \delta : \Theta \rightarrow \Delta}
    \hypo{\vdashG \nu : \Xi \rightarrow \Theta}
    \infer3{\vdashG (\sigma \delta)\nu = \sigma(\delta \nu) : \Xi \rightarrow \Gamma}
  }
  \and
  \ebrule{
    \hypo{\vdashG \sigma : \Gamma \rightarrow 1}
    \infer1{\vdashG \sigma = ()_\Gamma : \Gamma \rightarrow 1}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashG A \; \type}
    \infer1{\Gamma \vdashG A\Id_\Gamma = A}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashG A \; \type}
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \hypo{\vdashG \delta : \Theta \rightarrow \Delta}
    \infer3{\Theta \vdashG (A\sigma)\delta = A(\sigma\delta)}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashG a : A}
    \infer1{\Gamma \vdashG a\Id_\Gamma = a : A}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashG a : A}
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \hypo{\vdashG \delta : \Theta \rightarrow \Delta}
    \infer3{\Theta \vdashG (a\sigma)\delta = a(\sigma\delta) : A(\sigma\delta)}
  }
  \and
  \ebrule{
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \hypo{\Gamma \vdashG A \; \type}
    \hypo{\Delta \vdashG a : A\sigma}
    \infer3{\vdashG \p_{\Gamma,A}(\sigma, a) = \sigma : \Delta \rightarrow \Gamma}
  }
  \and
  \ebrule{
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \hypo{\Gamma \vdashG A \; \type}
    \hypo{\Delta \vdashG a : A\sigma}
    \infer3{\Delta \vdashG \q_{\Gamma,A}(\sigma, a) = a : A\sigma}
  }
  \and
  \ebrule{
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \hypo{\Gamma \vdashG A \; \type}
    \hypo{\Delta \vdashG a : A\sigma}
    \hypo{\vdashG \delta : \Theta \rightarrow \Delta}
    \infer4{\vdashG (\sigma, a)\delta = (\sigma\delta, a\delta) : \Theta \rightarrow \Gamma.A}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashG A \; \type}
    \infer1{\vdashG (\p_{\Gamma,A}, \q_{\Gamma,A}) = \Id_{\Gamma.A} : \Gamma.A \rightarrow \Gamma.A}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashG A \; \type}
    \hypo{\Gamma.A \vdashG B \; \type}
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \infer3{\Delta \vdashG \Pi(A,B)\sigma = \Pi(A\sigma,B\sigma^+)}
  }
  \and
  \ebrule{
    \hypo{\Gamma.A \vdashG b : B}
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \infer2{\Delta \vdashG \lambda(A,B,b)\sigma = \lambda(A\sigma,B\sigma^+,b\sigma^+) : \Pi(A\sigma,B\sigma^+)}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashG c : \Pi(A,B)}
    \hypo{\Gamma \vdashG a : A}
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \infer3{\Delta \vdashG \app(A,B,c,a)\sigma = \app(A\sigma,B\sigma^+,c\sigma,a\sigma) : B\sigma^+[a\sigma]}
  }
  \and
  \ebrule{
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \infer1{\Delta \vdashG (\U_n)_\Gamma\sigma = (\U_n)_\Delta}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashG a : (\U_n)_\Gamma}
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \infer2{\Delta \vdashG T_n(a)\sigma = T_n(a\sigma)}
  }
  \and
  \ebrule{
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \infer1[($m<n$)]{\Delta \vdashG (\U_m^n)_\Gamma\sigma = (\U_m^n)_\Delta : (\U_n)_\Delta}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashG a : (\U_m)_\Gamma}
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \infer2[($m\leq n$)]{\Delta \vdashG t_m^n(a)\sigma = t_m^n(a\sigma) : (\U_n)_\Delta}
  }
  \and
  \ebrule{
    \hypo{\Gamma \vdashG a : (\U_n)_\Gamma}
    \hypo{\Gamma.T_n(a) \vdashG b : (\U_n)_{\Gamma.T_n(a)}}
    \hypo{\vdashG \sigma : \Delta \rightarrow \Gamma}
    \infer3{\Delta \vdashG \Pi^n(a, b)\sigma = \Pi^n(a\sigma, b\sigma^+) : (\U_n)_\Delta}
  }
\end{mathparsmall}
